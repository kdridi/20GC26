## 🎨 STANDARDS QUALITÉ NON-NÉGOCIABLES

### Code

```cpp
// ✅ STYLE APPROUVÉ PAR TOUS LES AGENTS

namespace pong {  // Namespace par jeu

// Constantes : SCREAMING_SNAKE_CASE
inline constexpr auto MAX_ENTITIES = 1000;
inline constexpr float GRAVITY = 9.81F;  // F majuscule

// Types : PascalCase
struct Ball {
    // Membres : snake_case (optionnel) ou camelCase
    float x{400.0F};  // Valeurs par défaut explicites
    float y{300.0F};

    // Trailing return type TOUJOURS
    constexpr auto update(float dt) noexcept -> void {
        x += vx * dt;
        y += vy * dt;
    }

    // Spaceship operator pour comparaisons
    auto operator<=>(const Ball&) const = default;
};

// Functions : camelCase
auto updatePhysics(float dt) noexcept -> void;

} // namespace pong
```

### Tests

```cpp
// Pattern : TEST(Class, Should_Behavior_When_Condition)
TEST(Ball, Should_MoveRight_When_PositiveVelocity) {
    // Arrange - État initial clair
    pong::Ball ball{};
    ball.vx = 100.0F;

    // Act - Une seule action
    ball.update(0.1F);

    // Assert - Vérification précise
    EXPECT_FLOAT_EQ(ball.x, 410.0F);
}
```
